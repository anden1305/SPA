\section{S9 Reviewer stress-test rebuttals}
\begin{enumerate}
\item \textbf{Incremental over mcRBM.} Cross-lab holdout, temporal HMM prior, zero-shot staging without subject embeddings; physiology panels for comparability.
\item \textbf{NMI vs 97\% accuracy.} Rose 2026 label noise; unsupervised discovery vs mimicking single-rater bias.
\item \textbf{Needs AccuSleep calibration.} Holdout uses encoder + prior only; no subject embedding or 10-min manual baseline per mouse.
\item \textbf{$K{=}15$ meaningless.} K sweep + PSD/bout panels + expert exclusion of rare states.
\item \textbf{VAE+HMM not novel.} Fusion of decoder-only conditioning, warm sticky HMM--GMM, and MSSV ladder is the contribution; SVAE is generic precedent.
\end{enumerate}

\section{S10 Landscape 2023--2026}
\textbf{REST} (Wang et al., \emph{Sleep Advances} 2026): supervised Transformer, $\kappa\approx 0.87$---current temporal ceiling.
\textbf{WUCSS} (Cusinato et al., 2024): unsupervised light/deep NREM without population latent training.
\textbf{FASTER2} (Yamada et al., 2024): Gaussian HMM on engineered features.
\textbf{Rose et al.} (2026 bioRxiv): manual consensus limits cross-lab supervised DL on MSSV.
Our work sits in the unsupervised discovery quadrant with explicit zero-shot holdout protocol.
